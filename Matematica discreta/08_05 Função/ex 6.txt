from matplotlib import pyplot as plt

def g(d):
    return d * 2.50 +6

def d(g):
    return (g-6)/2.50

def resultado():
    X = range(1,50)
    Y = [g(k) for k in X]
    plt.plot(X,Y,color = 'green')
    plt.xlabel('Distancia percorrida')
    plt.ylabel('(R$)')
    plt.grid(True)
    plt.show()



resultado()
