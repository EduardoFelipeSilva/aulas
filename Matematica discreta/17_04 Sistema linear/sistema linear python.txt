import numpy as np
from scipy import linalg

#vendeu dia 1 3 camisetas, 2 bone e 3 par de meia e deu 330
#vendeu dia 2 4 camisetas, 5 bone e 1 par de meia e deu 330
#vendeu dia 3 2 camisetas, 1 bone e 0 par de meia e deu 330
n = np.array([[3, 2, 3], [4, 5, 1], [2, 1, 0]])
c = np.array([[330],[585], [175]])


s = linalg.solve(n, c)


camiseta = int(s[0][0])
bone = int(s[1][0])
meia = int(s[2][0])



c = int(input("quantas camisetas foi compradas: "))
b = int(input("quantos bone foi comprados: "))
m = int(input("quantas meias foi compradas: "))
compra = camiseta * c + bone * b + meia * m

if compra >= 100:
  total = compra * 0.10 + compra
  print(total)
else:
  total = compra
  print(total)
